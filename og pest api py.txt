@app.route("/enable_pest_monitor", methods=["POST"])
def enable_monitor():

    data = request.json

    sql = """INSERT INTO pest_monitoring(user_id,crop,location)
             VALUES(%s,%s,%s)"""

    cursor.execute(sql,(data["user_id"],data["crop"],data["location"]))
    db.commit()

    return jsonify({"message":"Monitoring enabled"})


# fetch alerts
@app.route("/get_pest_alerts/<user_id>", methods=["GET"])
def get_alerts(user_id):

    cursor.execute("""
    SELECT crop,pest_name,location,alert_message,created_at
    FROM pest_alerts
    WHERE user_id=%s
    ORDER BY created_at DESC
    """,(user_id,))

    rows = cursor.fetchall()

    alerts = []

    for r in rows:
        alerts.append({
            "crop":r[0],
            "pest":r[1],
            "location":r[2],
            "alert_message":r[3],
            "time":str(r[4])
        })

    return jsonify(alerts)

@app.route("/pest_predict", methods=["POST"])
def pest_predict():

    data = request.json

    crop = data["crop"]
    location = data["location"]
    temp = data["temp"]
    rain = data["rain"]
    user_id = data["user_id"]

    pest = None
    message = None

    # pest prediction rules
    if crop.lower() == "rice":
        if rain > 5 and temp > 28:
            pest = "Brown Plant Hopper"

    if crop.lower() == "tomato":
        if temp > 30 and rain < 2:
            pest = "Leaf Miner"

    if crop.lower() == "cotton":
        if temp > 32:
            pest = "Pink Bollworm"

    if crop.lower() == "wheat":
        if rain > 3:
            pest = "Army Worm"

    if pest:

        message = f"⚠ Pest Risk: {pest} may affect {crop} in {location}. Apply pesticide."

        cursor.execute("""
        INSERT INTO pest_alerts(user_id,crop,pest_name,location,message)
        VALUES(%s,%s,%s,%s,%s)
        """,(user_id,crop,pest,location,message))

        db.commit()

        return jsonify({"alert":message})

    return jsonify({"alert":"No pest risk"})
